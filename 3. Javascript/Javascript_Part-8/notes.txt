# reduce method
    reduces the array to a single value

    arr.reduce(reducer function with 2 variables for (accumulator, element));

    //accumulator concept 
    //array sum logic possible using this and a lot of array questions


# find max using reduce
    let result=arr.reduce((max,ele)=>{
        if(ele>max){
            return ele;
        }
        else{
            return max;
        }
    })

# Spread

    function func(...arr){

    }

# Spread (Array Literals)
    we can easily copy a array using spread & also add other elements, by passing each and every individual array element

    let newArr=[...arr];       //deep copy too
    let newArr=[...arr, 1,2,3]

# Rest
    --concept is directly opposite to that of spread   --- it takes a lot of individual elements and combine them

    allows a function to take an indefinite no. of arguments and bundle them in array

    (((((argument named collection)))))

    function sum(...args){   //args-- arguments     //we will collect items as a array
        return args.reduce( (sum,el)=> sum+el );        //.reduce ***
    }

    //if we want to take extra arguments, we can take it along with arguments  -- function sum(msg, ...args)


# Destructuring
    storing values of array into multiple variables

    let names=["tony","bruce","steve","peter"];
    let [winner, runnerup]=names;
    console.log(winner,runnerup);   //"tony" "bruce"   //assigning by index

    //we can also combine rest+Destructuring concepts --- let [winner, runnerup, ...others]=names;
